import numpy as np
from scipy.stats import qmc

def pack_circles(n: int) -> np.ndarray:
    """
    Pack n circles in a unit square to maximize sum of radii.
    
    Args:
        n: Number of circles to pack
    
    Returns:
        Numpy array of shape (n, 3) where each row is (x, y, radius)
        All values should be between 0 and 1
        Circles must not overlap
        
    Important: Set "all" random seeds to 2025, including the packages (such as scipy sub-packages) involving random seeds.
    """
    np.random.seed(2025)
    if n == 0:
        return np.zeros((0, 3))
    
    # Halton sequence initialization
    sampler = qmc.Halton(d=2, seed=2025)
    points = sampler.random(n)
    circles = np.zeros((n, 3))
    circles[:, :2] = points
    circles[:, 2] = np.minimum(np.minimum(circles[:, 0], 1 - circles[:, 0]),
                              np.minimum(circles[:, 1], 1 - circles[:, 1])) * 0.2
    
    # Deterministic initialization and local refinement
    for i in range(n):
        x, y = circles[i, :2]
        max_r = min(x, 1-x, y, 1-y)
        
        for _ in range(50):  # Bounded number of iterations to avoid infinite loops
            if np.random.random() < 0.9:
                step_size = 0.05 * (1 - i/200)**2
                new_x = np.clip(x + np.random.uniform(-step_size, step_size), 0, 1)
                new_y = np.clip(y + np.random.uniform(-step_size, step_size), 0, 1)
            else:
                step_size = 0.4 * (1 - i/200)**2
                new_x = np.clip(x + np.random.normal(0, 1) * step_size, 0, 1)
                new_y = np.clip(y + np.random.normal(0, 1) * step_size, 0, 1)
            
            current_r = min(new_x, 1-new_x, new_y, 1-new_y)
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt((circles[j,0]-new_x)**2 + (circles[j,1]-new_y)**2)
                current_r = min(current_r, dist - circles[j,2])
            
            if current_r > max_r:
                x, y = new_x, new_y
                max_r = current_r
    
    # Final all-pairs boundary and overlap repair
    for i in range(n):
        for j in range(i+1, n):
            dx, dy = circles[i, :2] - circles[j, :2]
            dist = np.sqrt(dx**2 + dy**2)
            max_radius = min(circles[i, 2], circles[j, 2])
            if dist < (max_radius * np.sqrt(2)):
                circles[i, 2] = max_radius
                break
    
    return circles
